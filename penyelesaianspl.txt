"""
Penyelesaian Sistem Persamaan Linear (SPL) 3 Variabel
Metode: Eliminasi Gauss dengan Pivoting Parsial + Substitusi Mundur

Bentuk SPL:
    a11 x + a12 y + a13 z = b1
    a21 x + a22 y + a23 z = b2
    a31 x + a32 y + a33 z = b3

Algoritma:
1. Bentuk matriks augmented [A | b] berukuran 3 x 4.
2. Untuk setiap kolom k = 0..1:
   a. Pivoting parsial: cari baris dengan |nilai| terbesar di kolom k
      (dari baris k ke bawah), lalu tukar dengan baris k.
   b. Jika pivot ~ 0, SPL tidak punya solusi tunggal.
   c. Untuk setiap baris i di bawah k, hitung faktor m = a[i][k] / a[k][k]
      lalu kurangkan: baris_i = baris_i - m * baris_k.
3. Cek elemen pivot terakhir a[2][2]; jika ~ 0, tidak ada solusi tunggal.
4. Substitusi mundur:
   z = b3' / a33'
   y = (b2' - a23' z) / a22'
   x = (b1' - a12' y - a13' z) / a11'
"""

EPS = 1e-12


def selesaikan_spl_3(A, b):
    """Menyelesaikan A x = b untuk A berukuran 3x3. Mengembalikan [x, y, z]."""
    n = 3
    # Matriks augmented (salinan agar input asli tidak berubah)
    M = [list(map(float, A[i])) + [float(b[i])] for i in range(n)]

    # Eliminasi maju dengan pivoting parsial
    for k in range(n - 1):
        pivot = max(range(k, n), key=lambda i: abs(M[i][k]))
        if abs(M[pivot][k]) < EPS:
            raise ValueError("SPL tidak memiliki solusi tunggal.")
        M[k], M[pivot] = M[pivot], M[k]

        for i in range(k + 1, n):
            m = M[i][k] / M[k][k]
            for j in range(k, n + 1):
                M[i][j] -= m * M[k][j]

    if abs(M[n - 1][n - 1]) < EPS:
        raise ValueError("SPL tidak memiliki solusi tunggal.")

    # Substitusi mundur
    x = [0.0] * n
    for i in range(n - 1, -1, -1):
        jumlah = sum(M[i][j] * x[j] for j in range(i + 1, n))
        x[i] = (M[i][n] - jumlah) / M[i][i]
    return x


def main():
    print("SPL 3 Variabel: a1 x + a2 y + a3 z = b")
    A, b = [], []
    for i in range(3):
        baris = input(f"Persamaan {i + 1} (masukkan a1 a2 a3 b dipisah spasi): ")
        nilai = list(map(float, baris.split()))
        if len(nilai) != 4:
            raise ValueError("Harus memasukkan 4 angka per persamaan.")
        A.append(nilai[:3])
        b.append(nilai[3])

    try:
        x, y, z = selesaikan_spl_3(A, b)
        print(f"\nx = {x:.6f}\ny = {y:.6f}\nz = {z:.6f}")
    except ValueError as e:
        print(f"\n{e}")


if __name__ == "__main__":
    main()
